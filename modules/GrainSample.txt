;@ requires GrainCore.txt
opcode GrainSample, 0, kkkkkkkkkkkkkkkkkkkkkkkkkkkkkkkko
    kDensity, kSize, kPosition, kPosJitter, kPitch, kPitchJitter, kWindow, kPan, kSpread, kFreeze, kLevel, kSeed, kSlot, kScan, kTimeJitter, kReverse, kDensityIn, kSizeIn, kPositionIn, kPosJitterIn, kPitchIn, kPitchJitterIn, kWindowIn, kPanIn, kSpreadIn, kFreezeIn, kLevelIn, kTimeJitterIn, kReverseIn, kScanIn, kOutL, kOutR, iDisplay xin
    aDry = 0
    kDensityCV zkr kDensityIn
    kDensity = kDensity + kDensityCV * 24
    kSizeCV zkr kSizeIn
    kSize = kSize + kSizeCV * 200
    kPositionCV zkr kPositionIn
    kPosition = kPosition / 127 + kPositionCV * 1
    kPosJitterCV zkr kPosJitterIn
    kPosJitter = kPosJitter / 127 + kPosJitterCV * 1
    kPitchCV zkr kPitchIn
    kPitch = kPitch + kPitchCV * 12
    kPitchJitterCV zkr kPitchJitterIn
    kPitchJitter = kPitchJitter + kPitchJitterCV * 12
    kWindowCV zkr kWindowIn
    kWindow = kWindow + kWindowCV * 2
    kPanCV zkr kPanIn
    kPan = kPan / 127 + kPanCV * 1
    kSpreadCV zkr kSpreadIn
    kSpread = kSpread / 127 + kSpreadCV * 1
    kFreezeCV zkr kFreezeIn
    kFreeze = kFreeze + kFreezeCV * 1
    kLevelCV zkr kLevelIn
    kLevel = kLevel + kLevelCV * 12
    kTimeJitterCV zkr kTimeJitterIn
    kTimeJitter = kTimeJitter / 100 + kTimeJitterCV * 1
    kReverseCV zkr kReverseIn
    kReverse = kReverse + kReverseCV * 1
    kScanCV zkr kScanIn
    kScan = kScan + kScanCV * 100
    kDensity limit kDensity, 0, 24
    kSize limit kSize, 2, 400
    kScan limit kScan, -100, 100
    kSlot = int(limit(kSlot, 1, 8) + 0.5) - 1
    kScanPhase init 0
    kPreviousScanSlot init -1
    if kSlot != kPreviousScanSlot then
        kScanPhase = 0
        kPreviousScanSlot = kSlot
    elseif kFreeze < 0.5 && kScan != 0 then
        kScanPhase = kScanPhase + kScan / (100 * kr)
        kScanPhase = kScanPhase - floor(kScanPhase)
    endif
    if kScanPhase != 0 then
        kPosition = kPosition + kScanPhase
        kPosition = kPosition - floor(kPosition)
    endif
    kTable = giGrainSampleTables[kSlot]
    kFrames = giGrainSampleFrames[kSlot]
    kRate = giGrainSampleRates[kSlot]
    aWetL, aWetR CMGrainCoreExtended aDry, 0, kTable, kFrames, kRate, kDensity, kSize, kPosition, kPosJitter, kPitch, kPitchJitter, kWindow, kPan, kSpread, kFreeze, kSeed, kTimeJitter, kReverse, 0, 0, 0, 0, 0, iDisplay
    kLevel limit kLevel, -24, 12
    kLevel = pow(10, kLevel / 20)
    aLevel interp kLevel
    aOutL limit aWetL * aLevel, -1, 1
    aOutR limit aWetR * aLevel, -1, 1
    if iDisplay > 0 then
        tablewkt kSlot + 1, 659, iDisplay
        CMPulvisOutputView aOutL, iDisplay
    endif
    if kOutL > 1 then
        zaw aOutL, kOutL
    endif
    if kOutR > 1 then
        zaw aOutR, kOutR
    endif
endop
